Finance MoneyLoop One Shot Bootstrap

Run this one line PowerShell command on Windows 11 to provision the finance silo. It writes all required files, sets up scheduled tasks, and runs verification. 

Set-StrictMode -Version Latest; $ErrorActionPreference = 'Stop'; $base = 'D:\\BrownEyeLake'; $silo = Join-Path $base 'silo_finance'; $shared = Join-Path $base 'shared'; $secrets = Join-Path $base 'secrets'; $venv = Join-Path $shared 'bin\\venv'; $services = Join-Path $shared 'bin\\services'; $scripts = Join-Path $shared 'bin\\scripts'; $index = Join-Path $shared 'index'; New-Item -ItemType Directory -Force -Path $base, $silo, $shared, $secrets, $venv, $services, $scripts, $index | Out-Null; $reqSecrets = @('stripe_secret', 'stripe_webhook_secret', 'discord_bot_token', 'discord_client_secret', 'discord_guild_id', 'discord_role_id', 'discord_post_channel_id', 'webhook_base_url'); foreach ($name in $reqSecrets) { $path = Join-Path $secrets ($name + '.sec'); if (-not (Test-Path $path)) { $secure = Read-Host -AsSecureString "Enter $name"; $secure | ConvertFrom-SecureString | Set-Content -Encoding Ascii -Path $path; } }; $cloudTokenPath = Join-Path $secrets 'cloudflared_tunnel_token.sec'; if (-not (Test-Path $cloudTokenPath)) { $cloudToken = Read-Host -AsSecureString 'Enter cloudflared_tunnel_token (optional, press Enter to skip)'; if ($cloudToken.Length -gt 0) { $cloudToken | ConvertFrom-SecureString | Set-Content -Encoding Ascii -Path $cloudTokenPath; } }; $discordClientId = Read-Host 'Enter discord_client_id'; $configPath = Join-Path $silo 'publish\\config.json'; $webhookBase = (Get-Content -Path (Join-Path $secrets 'webhook_base_url.sec') -Raw | ConvertTo-SecureString | ForEach-Object { [Runtime.InteropServices.Marshal]::PtrToStringAuto([Runtime.InteropServices.Marshal]::SecureStringToBSTR($_)) }); $guildId = (Get-Content -Path (Join-Path $secrets 'discord_guild_id.sec') -Raw | ConvertTo-SecureString | ForEach-Object { [Runtime.InteropServices.Marshal]::PtrToStringAuto([Runtime.InteropServices.Marshal]::SecureStringToBSTR($_)) }); $roleId = (Get-Content -Path (Join-Path $secrets 'discord_role_id.sec') -Raw | ConvertTo-SecureString | ForEach-Object { [Runtime.InteropServices.Marshal]::PtrToStringAuto([Runtime.InteropServices.Marshal]::SecureStringToBSTR($_)) }); $postChannelId = (Get-Content -Path (Join-Path $secrets 'discord_post_channel_id.sec') -Raw | ConvertTo-SecureString | ForEach-Object { [Runtime.InteropServices.Marshal]::PtrToStringAuto([Runtime.InteropServices.Marshal]::SecureStringToBSTR($_)) }); $config = @{ currency = 'NZD'; price_cents = 100; stripe_success_url = ($webhookBase.TrimEnd('/') + '/success'); stripe_cancel_url = ($webhookBase.TrimEnd('/') + '/cancel'); webhook_base_url = $webhookBase; discord_guild_id = $guildId; discord_role_id = $roleId; discord_post_channel_id = $postChannelId; discord_client_id = $discordClientId }; $config | ConvertTo-Json -Depth 4 | Set-Content -Encoding Ascii -Path $configPath; if (-not (Test-Path $venv)) { python -m venv $venv }; & (Join-Path $venv 'Scripts\\python.exe') -m pip install --upgrade pip; & (Join-Path $venv 'Scripts\\python.exe') -m pip install flask stripe requests discord.py python-dotenv pywin32; if (-not (Get-Command cloudflared -ErrorAction SilentlyContinue) -and (Test-Path $cloudTokenPath)) { winget install --id Cloudflare.cloudflared -e --accept-package-agreements --accept-source-agreements }; & (Join-Path $scripts 'verify.ps1'); Write-Host 'Bootstrap complete';

This command is safe to rerun. Secrets are stored with DPAPI and will not be printed.
