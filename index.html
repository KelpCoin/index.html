def db() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH)
    conn.execute("PRAGMA journal_mode=WAL;")
    conn.execute("PRAGMA synchronous=NORMAL;")
    return conn

def init_db() -> None:
    conn = db()
    conn.execute(
        "CREATE TABLE IF NOT EXISTS events (id INTEGER PRIMARY KEY AUTOINCREMENT, ts TEXT, tenant_id TEXT, kind TEXT, detail TEXT)"
    )
    conn.execute(
        "CREATE TABLE IF NOT EXISTS email_map (email TEXT PRIMARY KEY, discord_user_id TEXT)"
    )
    conn.commit()
    conn.close()


def log_event(tenant_id: str, kind: str, detail: str) -> None:
    conn = db()
    conn.execute(
        "INSERT INTO events (ts, tenant_id, kind, detail) VALUES (?, ?, ?, ?)",
        (datetime.datetime.utcnow().isoformat() + "Z", tenant_id, kind, detail),
    )
    conn.commit()
    conn.close()


def load_tenants() -> Dict[str, Dict[str, Any]]:
    tenants = {}
    root = Path(TENANTS_ROOT)
    for tenant_dir in root.iterdir():
        if tenant_dir.is_dir():
            tenant_path = tenant_dir / "tenant.json"
            if tenant_path.exists():
                with open(tenant_path, "r", encoding="ascii") as f:
                    tenant = json.load(f)
                tenants[tenant["tenant_id"]] = tenant
    return tenants


def get_email_mapping(email: str) -> str:
    conn = db()
    cur = conn.execute("SELECT discord_user_id FROM email_map WHERE email = ?", (email,))
    row = cur.fetchone()
    conn.close()
    return row[0] if row else ""


def update_daily_progress(artifact_path: str, result: str) -> None:
    today = datetime.datetime.utcnow().strftime("%Y%m%d")
    day_dir = Path(DAILY_ROOT) / "days" / today
    day_dir.mkdir(parents=True, exist_ok=True)
    log_path = day_dir / f"snapshot_{today}.log"
    line = f"{datetime.datetime.utcnow().isoformat()}Z | {artifact_path} | {result}\n"
    with open(log_path, "a", encoding="ascii") as f:
        f.write(line)
    checkpoint_path = Path(DAILY_ROOT) / "checkpoint.json"
    payload = {
        "last_artifact": artifact_path,
        "last_result": result,
        "last_updated": datetime.datetime.utcnow().isoformat() + "Z",
    }
    with open(checkpoint_path, "w", encoding="ascii") as f:
        json.dump(payload, f)


def write_job(tenant_id: str, payload: Dict[str, Any]) -> str:
    queue_dir = Path(TENANTS_ROOT) / tenant_id / "queue"
    queue_dir.mkdir(parents=True, exist_ok=True)
    job_id = f"job_{int(time.time())}_{hashlib.sha256(json.dumps(payload).encode('ascii')).hexdigest()[:8]}"
    job_path = queue_dir / f"{job_id}.json"
    with open(job_path, "w", encoding="ascii") as f:
        json.dump(payload, f)
    return str(job_path)


def process_job(job_path: Path) -> str:
    with open(job_path, "r", encoding="ascii") as f:
        job = json.load(f)
    tenant_id = job["tenant_id"]
    offer_id = job["offer_id"]
    email = job.get("email", "")
    discord_user_id = get_email_mapping(email) if email else ""
    artifact_name = f"proof_{tenant_id}_{offer_id}_{int(time.time())}.txt"
    artifact_path = Path(OUTBOX_PATH) / artifact_name
    lines = [
        "PROOF ARTIFACT",
        f"tenant_id={tenant_id}",
        f"offer_id={offer_id}",
        f"email={email}",
        f"discord_user_id={discord_user_id}",
        f"timestamp={datetime.datetime.utcnow().isoformat()}Z",
    ]
    with open(artifact_path, "w", encoding="ascii") as f:
        f.write("\n".join(lines) + "\n")

    result = "fulfilled"
    if DISCORD_BOT_TOKEN and discord_user_id:
        try:
            tenant = load_tenants()[tenant_id]
            guild_id = tenant.get("discord", {}).get("guild_id", "")
            role_id = tenant.get("discord", {}).get("paid_role_id", "")
            proof_channel_id = tenant.get("discord", {}).get("proof_channel_id", "")
            headers = {"Authorization": f"Bot {DISCORD_BOT_TOKEN}", "Content-Type": "application/json"}
            if guild_id and role_id:
                requests.put(
                    f"https://discord.com/api/guilds/{guild_id}/members/{discord_user_id}/roles/{role_id}",
                    headers=headers,
                    timeout=10,
                )
            if proof_channel_id:
                requests.post(
                    f"https://discord.com/api/channels/{proof_channel_id}/messages",
                    headers=headers,
                    json={"content": f"Proof {artifact_name}"},
                    timeout=10,
                )
        except Exception as exc:
            result = f"discord_error:{exc}"

    update_daily_progress(str(artifact_path), result)
    log_event(tenant_id, "fulfilled", json.dumps({"artifact": str(artifact_path), "result": result}))
    job_path.unlink(missing_ok=True)
    return result

@app.route("/dash")
def dash():
    tenant_filter = request.args.get("tenant", "")
    conn = db()
    if tenant_filter:
        cur = conn.execute(
            "SELECT ts, tenant_id, kind, detail FROM events WHERE tenant_id = ? ORDER BY id DESC LIMIT 50",
            (tenant_filter,),
        )
    else:
        cur = conn.execute("SELECT ts, tenant_id, kind, detail FROM events ORDER BY id DESC LIMIT 50")
    rows = cur.fetchall()
    conn.close()
    lines = [" | ".join(row) for row in rows]
    return render_template_string(DASH_TEMPLATE, tenant_filter=tenant_filter, events_text="\n".join(lines))


@app.route("/link", methods=["GET", "POST"])
def link():
    status = ""
    if request.method == "POST":
        email = request.form.get("email", "").strip().lower()
        discord_user_id = request.form.get("discord_user_id", "").strip()
        if email and discord_user_id:
            conn = db()
            conn.execute("INSERT OR REPLACE INTO email_map (email, discord_user_id) VALUES (?, ?)", (email, discord_user_id))
            conn.commit()
            conn.close()
            status = "saved"
        else:
            status = "missing_fields"
    return render_template_string(LINK_TEMPLATE, status=status)

        return "unknown tenant", 404

@app.route("/buy")
def buy():
    tenant_id = request.args.get("tenant_id", "")
    offer_id = request.args.get("offer_id", "")
    tenants = load_tenants()
    if tenant_id not in tenants:
        return "unknown tenant", 404
    tenant = tenants[tenant_id]
    price_id = tenant.get("universal_price_id", "") or ""
    if not price_id:
        return "missing price id", 400
    offer_hash = hashlib.sha256(f"{tenant_id}:{offer_id}".encode("ascii")).hexdigest()
    session = stripe.checkout.Session.create(
        mode="payment",
        line_items=[{"price": price_id, "quantity": 1}],
        success_url="http://127.0.0.1:{0}/dash".format(CONFIG["port"]),
        cancel_url="http://127.0.0.1:{0}/dash".format(CONFIG["port"]),
        metadata={
            "tenant_id": tenant_id,
            "offer_id": offer_id,
            "silo_tag": tenant.get("silo_tag", ""),
            "offer_hash": offer_hash,
        },
    )
    return redirect(session.url)


@app.route("/webhook", methods=["POST"])
def webhook():
    payload = request.data
    sig_header = request.headers.get("Stripe-Signature", "")
    try:
        event = stripe.Webhook.construct_event(payload=payload, sig_header=sig_header, secret=STRIPE_WEBHOOK_SECRET)
    except Exception:
        return "invalid", 400

    if event["type"] == "checkout.session.completed":
        data = event["data"]["object"]
        tenant_id = data.get("metadata", {}).get("tenant_id", "")
        offer_id = data.get("metadata", {}).get("offer_id", "")
        email = data.get("customer_details", {}).get("email", "")
        job = {"tenant_id": tenant_id, "offer_id": offer_id, "email": email}
        write_job(tenant_id, job)
        log_event(tenant_id, "webhook", json.dumps(job))
    return "ok", 200


@app.route("/work_once", methods=["POST", "GET"])
def work_once():
    if ADMIN_KEY and request.headers.get("x-admin-key", "") != ADMIN_KEY:
        return "forbidden", 403
    processed = 0
    tenants = load_tenants()
    for tenant_id in tenants:
        queue_dir = Path(TENANTS_ROOT) / tenant_id / "queue"
        for job_path in list(queue_dir.glob("*.json")):
            process_job(job_path)
            processed += 1
            if processed >= 10:
                break
    return jsonify({"processed": processed})


@app.route("/dev_simulate", methods=["POST"])
def dev_simulate():
    if ADMIN_KEY and request.headers.get("x-admin-key", "") != ADMIN_KEY:
        return "forbidden", 403
    payload = request.get_json(force=True)
    tenant_id = payload.get("tenant_id", "")
    offer_id = payload.get("offer_id", "")
    email = payload.get("email", "")
    job = {"tenant_id": tenant_id, "offer_id": offer_id, "email": email}
    write_job(tenant_id, job)
    log_event(tenant_id, "dev_simulate", json.dumps(job))
    return jsonify({"queued": True})
