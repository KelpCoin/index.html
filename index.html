        "powershell",
        "-NoProfile",
        "-Command",
        f"Get-Content -Path '{path}' | ConvertTo-SecureString | "
        f"ForEach-Object {{ [System.Net.NetworkCredential]::new('', $_).Password }}",

import json
import os
import stripe
from lake_lib import read_secret, read_config, log_line, upsert_event
from revenue_sources import SOURCES


def slugify(text):
    return "".join(ch.lower() if ch.isalnum() else "-" for ch in text).strip("-")


def main(dry_run=False):
    cfg = read_config()
    key = read_secret("stripe_secret")
    if not key:
        log_line("stripe_links", "Stripe secret missing")
        if dry_run:
            print("DRY-RUN: stripe key missing")
            return
        raise SystemExit("Stripe secret missing")
    stripe.api_key = key

    created = 0
    for idx, name in enumerate(SOURCES, start=1):
        sku_id = f"FIN-{idx:03d}"
        slug = slugify(name)
        if dry_run:
            print(f"DRY-RUN: would create {sku_id} {slug}")
            continue
        product = stripe.Product.create(name=f"{sku_id} {name}", metadata={"sku_id": sku_id})
        price = stripe.Price.create(
            product=product.id,
            unit_amount=int(cfg.get("price_cents", 100)),
            currency=cfg.get("currency", "nzd")
        )
        link = stripe.PaymentLink.create(
            line_items=[{"price": price.id, "quantity": 1}],
            after_completion={
                "type": "redirect",
                "redirect": {"url": cfg["stripe_success_url"]}
            },
            metadata={"sku_id": sku_id, "slug": slug}
        )
        upsert_event("stripe_link", {"sku_id": sku_id, "link": link.url})
        created += 1
    log_line("stripe_links", f"Created {created} links")


if __name__ == "__main__":
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--dry-run", action="store_true")
    args = ap.parse_args()
    main(dry_run=args.dry_run)

import json

from flask import Flask, request, redirect
import requests
import stripe
from lake_lib import read_secret, read_config, ensure_db, INDEX_DB, log_line, upsert_event

app = Flask(__name__)


def db_conn():
    ensure_db()
    return sqlite3.connect(INDEX_DB)


def store_claim(stripe_customer_id):
    token = secrets.token_urlsafe(24)
    conn = db_conn()
    cur = conn.cursor()
    cur.execute(
        "INSERT OR REPLACE INTO claims (claim_token, stripe_customer_id, created_utc, used) VALUES (?, ?, datetime('now'), 0)",
        (token, stripe_customer_id),
    )
    conn.commit()
    conn.close()
    return token


def mark_claim_used(token):
    conn = db_conn()
    cur = conn.cursor()
    cur.execute("UPDATE claims SET used = 1 WHERE claim_token = ?", (token,))
    conn.commit()
    conn.close()


def get_customer_for_claim(token):
    conn = db_conn()
    cur = conn.cursor()
    cur.execute("SELECT stripe_customer_id, used FROM claims WHERE claim_token = ?", (token,))
    row = cur.fetchone()
    conn.close()
    if not row:
        return "", True
    return row[0], bool(row[1])


def set_identity(stripe_customer_id, discord_user_id):
    conn = db_conn()
    cur = conn.cursor()
    cur.execute(
        "INSERT OR REPLACE INTO identity_map (stripe_customer_id, discord_user_id, created_utc) VALUES (?, ?, datetime('now'))",
        (stripe_customer_id, discord_user_id),
    )
    conn.commit()
    conn.close()


def assign_role(discord_user_id, guild_id, role_id, bot_token):
    url = f"https://discord.com/api/v10/guilds/{guild_id}/members/{discord_user_id}/roles/{role_id}"
    headers = {"Authorization": f"Bot {bot_token}", "Content-Type": "application/json"}
    resp = requests.put(url, headers=headers)
    return resp.status_code in (200, 204)


@app.route("/health", methods=["GET"])
def health():
    return "ok", 200


@app.route("/stripe/webhook", methods=["POST"])
def stripe_webhook():
    payload = request.data
    sig = request.headers.get("Stripe-Signature", "")
    whsec = read_secret("stripe_webhook_secret")
    if not whsec:
        return "missing webhook secret", 400
    try:
        event = stripe.Webhook.construct_event(payload, sig, whsec)
    except Exception as exc:
        log_line("webhook", f"stripe verify failed {exc}")
        return "bad signature", 400

    if event["type"] == "checkout.session.completed":
        session = event["data"]["object"]
        customer_id = session.get("customer", "")
        if customer_id:
            token = store_claim(customer_id)
            cfg = read_config()
            claim_url = f"{cfg['webhook_base_url']}/claim?token={token}"
            upsert_event("claim_created", {"customer_id": customer_id, "claim_url": claim_url})
            log_line("webhook", f"claim link {claim_url}")
    return "ok", 200


@app.route("/claim", methods=["GET"])
def claim():
    token = request.args.get("token", "")
    customer_id, used = get_customer_for_claim(token)
    if not customer_id or used:
        return "invalid claim", 400
    cfg = read_config()
    redirect_uri = f"{cfg['webhook_base_url']}/oauth/callback"
    client_id = cfg["discord_client_id"]
    auth_url = (
        "https://discord.com/api/oauth2/authorize"
        f"?client_id={client_id}"
        f"&redirect_uri={redirect_uri}"
        "&response_type=code&scope=identify"
        f"&state={token}"
    )
    return redirect(auth_url)


@app.route("/oauth/callback", methods=["GET"])
def oauth_callback():
    code = request.args.get("code", "")
    state = request.args.get("state", "")
    customer_id, used = get_customer_for_claim(state)
    if not customer_id or used:
        return "invalid claim", 400
    cfg = read_config()
    redirect_uri = f"{cfg['webhook_base_url']}/oauth/callback"
    data = {
        "client_id": cfg["discord_client_id"],
        "client_secret": read_secret("discord_client_secret"),
        "grant_type": "authorization_code",
        "code": code,
        "redirect_uri": redirect_uri,
        "scope": "identify",
    }
    headers = {"Content-Type": "application/x-www-form-urlencoded"}
    token_resp = requests.post("https://discord.com/api/oauth2/token", data=data, headers=headers)
    if token_resp.status_code != 200:
        return "oauth failed", 400
    access_token = token_resp.json().get("access_token", "")
    user_resp = requests.get("https://discord.com/api/users/@me", headers={"Authorization": f"Bearer {access_token}"})
    if user_resp.status_code != 200:
        return "user fetch failed", 400
    user_id = user_resp.json().get("id", "")
    if not user_id:
        return "user missing", 400
    set_identity(customer_id, user_id)
    mark_claim_used(state)
    bot_token = read_secret("discord_bot_token")
    ok = assign_role(user_id, cfg["discord_guild_id"], cfg["discord_default_role_id"], bot_token)
    upsert_event("claim_bound", {"stripe_customer_id": customer_id, "discord_user_id": user_id, "role_assigned": ok})
    return "claim complete", 200


if __name__ == "__main__":
    stripe.api_key = read_secret("stripe_secret")

import json
import os
import sqlite3
from datetime import datetime
from lake_lib import INDEX_DB, ensure_db, log_line, upsert_event, upsert_table

BASE = r"D:\\BrownEyeLake"
TELEMETRY_DIR = os.path.join(BASE, "silo_finance", "telemetry")
PRODUCTS_DIR = os.path.join(BASE, "silo_finance", "products")


def compute_sales():
    ensure_db()
    conn = sqlite3.connect(INDEX_DB)
    cur = conn.cursor()
    cur.execute("SELECT payload FROM events WHERE event_type = 'sale'")
    sales = [json.loads(row[0]) for row in cur.fetchall()]
    conn.close()
    counts = {}
    for item in sales:
        sku = item.get("sku_id", "")
        counts[sku] = counts.get(sku, 0) + 1
    return counts


def mark_cooldown(sales_counts):
    for name in os.listdir(PRODUCTS_DIR):
        if not name.endswith(".json"):
            continue
        path = os.path.join(PRODUCTS_DIR, name)
        with open(path, "r", encoding="ascii") as f:
            product = json.load(f)
        sku = product.get("sku_id", "")
        if sales_counts.get(sku, 0) == 0:
            product["status"] = "cooldown"
        else:
            product["status"] = "active"
        with open(path, "w", encoding="ascii") as f:
            json.dump(product, f, indent=2)


def create_variants(sales_counts):
    winners = sorted(sales_counts.items(), key=lambda x: x[1], reverse=True)[:3]
    for sku, count in winners:
        if count <= 0:
            continue
        source_path = ""
        for name in os.listdir(PRODUCTS_DIR):
            if not name.endswith(".json"):
                continue
            path = os.path.join(PRODUCTS_DIR, name)
            with open(path, "r", encoding="ascii") as f:
                product = json.load(f)
            if product.get("sku_id") == sku and not product.get("variant_of"):
                source_path = path
                base_product = product
                break
        if not source_path:
            continue
        for suffix in ["A", "B"]:
            variant_sku = f"{sku}-{suffix}"
            variant_file = os.path.join(PRODUCTS_DIR, f"{variant_sku}.json")
            if os.path.exists(variant_file):
                continue
            variant = dict(base_product)
            variant["id"] = f"product-{variant_sku}"
            variant["sku_id"] = variant_sku
            variant["variant_of"] = sku
            variant["copy_variant"] = suffix
            with open(variant_file, "w", encoding="ascii") as f:
                json.dump(variant, f, indent=2)
            upsert_table("products", variant)
        log_line("telemetry", f"Variants generated for {sku}")
        upsert_event("variant_generated", {"sku_id": sku})


def main():
    os.makedirs(TELEMETRY_DIR, exist_ok=True)
    sales = compute_sales()
    output = {
        "created_utc": datetime.utcnow().isoformat(),
        "sales": sales,
        "low_performers": [k for k, v in sales.items() if v == 0]
    }
    out_path = os.path.join(TELEMETRY_DIR, "daily_telemetry.json")
    with open(out_path, "w", encoding="ascii") as f:
        json.dump(output, f, indent=2)
    mark_cooldown(sales)
    create_variants(sales)
    log_line("telemetry", f"Telemetry computed for {len(sales)} skus")
    upsert_event("telemetry", output)


if __name__ == "__main__":
    main()
