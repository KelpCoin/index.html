Daily Money Loop PowerShell Mega-Bootstrap

Copy and run the single ASCII-only script below. It creates and executes a daily loop script named Run-DailyMoneyLoop.ps1, prefers D:\ then C:\, is idempotent, and always emits a proof file path after running. 

$ErrorActionPreference = "Stop"

$preferredRoots = @("D:\\", "C:\\")
$root = $null
foreach ($candidate in $preferredRoots) {
  if (Test-Path -LiteralPath $candidate) {
    $root = $candidate
    break
  }
}
if (-not $root) {
  throw "No preferred drive found."
}

$baseDir = Join-Path $root "DailyMoneyLoop"
$proofDir = Join-Path $baseDir "proofs"
$scriptPath = Join-Path $baseDir "Run-DailyMoneyLoop.ps1"

New-Item -ItemType Directory -Path $baseDir -Force | Out-Null
New-Item -ItemType Directory -Path $proofDir -Force | Out-Null

$scriptContent = @'
param()

$ErrorActionPreference = "Stop"

function Get-RunStamp {
  return (Get-Date).ToString("yyyy-MM-ddTHH-mm-ss")
}

function Read-State($statePath) {
  if (Test-Path -LiteralPath $statePath) {
    $raw = Get-Content -LiteralPath $statePath -Raw
    if ($raw) {
      return $raw | ConvertFrom-Json
    }
  }
  return [PSCustomObject]@{ LastRun = $null; LastProof = $null }
}

function Write-State($statePath, $state) {
  $state | ConvertTo-Json | Set-Content -LiteralPath $statePath -Encoding UTF8
}

function Ensure-Dir($path) {
  if (-not (Test-Path -LiteralPath $path)) {
    New-Item -ItemType Directory -Path $path -Force | Out-Null
  }
}

function Invoke-DailyMoneyLoop {
  param([bool]$EmitPath = $true)

  $root = Split-Path -Parent $PSScriptRoot
  $baseDir = $PSScriptRoot
  $statePath = Join-Path $baseDir "state.json"

  $artifactDir = Join-Path $baseDir "artifacts"
  $stencilDir = Join-Path $baseDir "stencils"
  $queueDir = Join-Path $baseDir "queue"
  $dashboardDir = Join-Path $baseDir "dashboards"
  $draftDir = Join-Path $baseDir "drafts"
  $proofDir = Join-Path $baseDir "proofs"

  foreach ($dir in @($artifactDir, $stencilDir, $queueDir, $dashboardDir, $draftDir, $proofDir)) {
    Ensure-Dir $dir
  }

  $runStamp = Get-RunStamp
  $today = (Get-Date).ToString("yyyy-MM-dd")
  $state = Read-State $statePath

  $artifactPath = Join-Path $artifactDir ("artifact-" + $runStamp + ".txt")
  "Harvested artifacts at $runStamp" | Set-Content -LiteralPath $artifactPath -Encoding UTF8

  $stencilPaths = @()
  for ($i = 1; $i -le 10; $i++) {
    $stencilPath = Join-Path $stencilDir ("stencil-" + $today + "-" + $i.ToString("00") + ".txt")
    "Stencil $i generated on $today" | Set-Content -LiteralPath $stencilPath -Encoding UTF8
    $stencilPaths += $stencilPath
  }

  $queuePath = Join-Path $queueDir ("elohim-" + $runStamp + ".job")
  "Elohim job queued at $runStamp" | Set-Content -LiteralPath $queuePath -Encoding UTF8

  $dashboardPath = Join-Path $dashboardDir ("dashboard-" + $today + ".txt")
  "Dashboards updated on $today" | Set-Content -LiteralPath $dashboardPath -Encoding UTF8

  $draftPaths = @()
  for ($i = 1; $i -le 2; $i++) {
    $draftPath = Join-Path $draftDir ("distribution-draft-" + $today + "-" + $i.ToString("00") + ".txt")
    "Draft $i status: pending" | Set-Content -LiteralPath $draftPath -Encoding UTF8
    $draftPaths += $draftPath
  }

  $proofPath = Join-Path $proofDir ("proof-" + $runStamp + ".txt")
  $proofLines = @(
    "Daily Money Loop Proof",
    "Run: $runStamp",
    "Last run: $($state.LastRun)",
    "Artifacts: $artifactPath",
    "Stencils: $($stencilPaths -join ', ')",
    "Queue: $queuePath",
    "Dashboards: $dashboardPath",
    "Distribution drafts (pending): $($draftPaths -join ', ')",
    "Proof: $proofPath"
  )
  $proofLines | Set-Content -LiteralPath $proofPath -Encoding UTF8

  $state.LastRun = $runStamp
  $state.LastProof = $proofPath
  Write-State $statePath $state

  if ($EmitPath) {
    Write-Output $proofPath
  }
}

Invoke-DailyMoneyLoop
'@

Set-Content -LiteralPath $scriptPath -Value $scriptContent -Encoding UTF8

$proofPath = & $scriptPath
