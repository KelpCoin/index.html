function Confirm-GpuProven {
  $proofPath = "C:\BrownEyeCortex\SignalLayer\GPU\GPU_PROVEN.json"
  if (-not (Test-Path -LiteralPath $proofPath)) {
    Write-GpuGateLog "GPU proof missing at $proofPath"
    return $false
  }
  try {
    $raw = Get-Content -LiteralPath $proofPath -Raw -ErrorAction Stop
    $data = $raw | ConvertFrom-Json
    if (-not $data.gpu_proven) {
      Write-GpuGateLog "GPU proof present but gpu_proven is false"
      return $false
    }
    $testedAt = [DateTime]::Parse($data.tested_at).ToUniversalTime()
    $fresh = (Get-Date).ToUniversalTime().AddDays(-7)
    if ($testedAt -lt $fresh) {
      Write-GpuGateLog "GPU proof stale: tested_at $($data.tested_at)"
      return $false
    }
    Write-GpuGateLog "GPU proof valid: method $($data.method) tested_at $($data.tested_at)"
    return $true
  } catch {
    Write-GpuGateLog "GPU proof parse error: $($_.Exception.Message)"
    return $false
  }
}

    continue

  Write-ExecutorLog "Executing $($task.name)"
  $executed += $task.name
  $startInfo = @{

    ArgumentList = "-NoProfile -ExecutionPolicy Bypass -Command", $task.command

    PassThru = $true
    WindowStyle = "Hidden"

    $tstate.next_allowed_at = (Get-Date).AddSeconds($cooldown).ToString("s")
    Write-ExecutorLog "Task $($task.name) timed out, cooldown $cooldown seconds"
