PowerShell Mega-Bootstrap

Copy the script below into PowerShell. It creates a human-approval gate, a draft generator, and a scheduler that only moves approved drafts into a ready-to-send folder. 

#requires -version 5.1
$ErrorActionPreference = "Stop"

function Get-RootPath {
  $preferred = "D:\\BrownEyeCortex"
  if (Test-Path -LiteralPath "D:\\") { return $preferred }
  return "C:\\BrownEyeCortex"
}

$root = Get-RootPath
$gate = Join-Path $root "PublishGate"
$pending = Join-Path $gate "pending"
$approved = Join-Path $gate "approved"
$rejected = Join-Path $gate "rejected"
$ready = Join-Path $gate "ready_to_send"
$secrets = Join-Path $gate "secrets"

$paths = @($root, $gate, $pending, $approved, $rejected, $ready, $secrets)
foreach ($path in $paths) {
  if (-not (Test-Path -LiteralPath $path)) {
    New-Item -Path $path -ItemType Directory | Out-Null
  }
}

function Set-EncryptedWebhook {
  param(
    [Parameter(Mandatory = $true)]
    [string]$Name,
    [Parameter(Mandatory = $true)]
    [string]$PlaintextWebhook
  )

  $secure = ConvertTo-SecureString -String $PlaintextWebhook -AsPlainText -Force
  $cipher = $secure | ConvertFrom-SecureString
  $target = Join-Path $secrets ("{0}.dpapi" -f $Name)
  Set-Content -LiteralPath $target -Value $cipher -Encoding ASCII

  Write-Host "Encrypted webhook saved to $target"
  Write-Host "Rotate immediately if exposed. Never store plaintext webhooks in files."
}

function Get-EncryptedWebhook {
  param(
    [Parameter(Mandatory = $true)]
    [string]$Name
  )

  $target = Join-Path $secrets ("{0}.dpapi" -f $Name)
  if (-not (Test-Path -LiteralPath $target)) {
    throw "Encrypted webhook not found: $target"
  }
  $cipher = Get-Content -LiteralPath $target -Raw
  $secure = $cipher | ConvertTo-SecureString
  return [System.Net.NetworkCredential]::new("", $secure).Password
}

function New-DraftFile {
  param(
    [Parameter(Mandatory = $true)]
    [string]$Silo,
    [Parameter(Mandatory = $true)]
    [string]$PostText,
    [string]$FilenameTag
  )

  $timestamp = Get-Date -Format "yyyyMMdd_HHmmss"
  $tag = if ($FilenameTag) { $FilenameTag } else { $Silo }
  $file = Join-Path $pending ("draft_{0}_{1}.txt" -f $timestamp, $tag)

  $lines = @(
    "silo: $Silo",
    "post_text:",
    $PostText,
    "link_placeholders:",
    "wise_pay_link: https://wise.com/pay/me/joshuaa495",
    "patreon_optional_for_mtg: https://www.patreon.com/HappyHomaroid"
  )

  Set-Content -LiteralPath $file -Value $lines -Encoding UTF8
  return $file
}

function New-PendingDrafts {
  param(
    [int]$Count = 5
  )

  $silos = @(
    @{ name = "MTG"; prompt = "Share a deck insight and invite support." },
    @{ name = "Fitness"; prompt = "Share a training tip and value proposition." },
    @{ name = "Tech"; prompt = "Share a build update and offer value." },
    @{ name = "Art"; prompt = "Share a progress snippet and invite feedback." },
    @{ name = "Business"; prompt = "Share a small win and mention consulting." }
  )

  for ($i = 0; $i -lt $Count; $i++) {
    $silo = $silos[$i % $silos.Count]
    $patreon = if ($silo.name -eq "MTG") { " Optional Patreon mention: https://www.patreon.com/HappyHomaroid" } else { "" }
    $text = "" +
      $silo.prompt +
      " Include a Wise pay link placeholder. Keep it helpful and non-pushy." +
      $patreon

    New-DraftFile -Silo $silo.name -PostText $text -FilenameTag ("seed{0}" -f ($i + 1)) | Out-Null
    Start-Sleep -Milliseconds 200
  }
}

function Show-ApprovalMechanics {
  Write-Host "Approval mechanics:"
  Write-Host "1) Drafts are created in: $pending"
  Write-Host "2) Human review moves a draft into: $approved"
  Write-Host "3) Create an approval file with the same name plus .approve extension"
  Write-Host "   Example: draft_20240101_120000_seed1.txt.approve"
  Write-Host "4) Scheduler moves approved + approval file into: $ready"
  Write-Host "5) Rejected drafts go to: $rejected"
  Write-Host "No posting occurs in this script. Ready drafts are only staged."
}

function Move-ApprovedToReady {
  $approvedFiles = Get-ChildItem -LiteralPath $approved -File -Filter "*.txt" -ErrorAction SilentlyContinue
  foreach ($file in $approvedFiles) {
    $approvalFile = "$($file.FullName).approve"
    if (Test-Path -LiteralPath $approvalFile) {
      Move-Item -LiteralPath $file.FullName -Destination $ready
      Move-Item -LiteralPath $approvalFile -Destination $ready
    }
  }
}

$taskName = "BrownEyeCortex_ApprovalGate"
$taskExists = Get-ScheduledTask -TaskName $taskName -ErrorAction SilentlyContinue
$moveCommand = @'
$root = if (Test-Path -LiteralPath "D:\") { "D:\BrownEyeCortex" } else { "C:\BrownEyeCortex" }
$approved = Join-Path $root "PublishGate\approved"
$ready = Join-Path $root "PublishGate\ready_to_send"
if (Test-Path -LiteralPath $approved) {
  $approvedFiles = Get-ChildItem -LiteralPath $approved -File -Filter "*.txt" -ErrorAction SilentlyContinue
  foreach ($file in $approvedFiles) {
    $approvalFile = "$($file.FullName).approve"
    if (Test-Path -LiteralPath $approvalFile) {
      Move-Item -LiteralPath $file.FullName -Destination $ready
      Move-Item -LiteralPath $approvalFile -Destination $ready
    }
  }
}
'@
$taskAction = New-ScheduledTaskAction -Execute "PowerShell.exe" -Argument "-NoProfile -WindowStyle Hidden -Command $moveCommand"

$taskTriggers = @(
  New-ScheduledTaskTrigger -Daily -At 06:00,
  New-ScheduledTaskTrigger -Daily -At 11:30,
  New-ScheduledTaskTrigger -Daily -At 17:00,
  New-ScheduledTaskTrigger -Daily -At 21:30
)

if (-not $taskExists) {
  $principal = New-ScheduledTaskPrincipal -UserId $env:USERNAME -LogonType Interactive -RunLevel LeastPrivilege
  Register-ScheduledTask -TaskName $taskName -Action $taskAction -Trigger $taskTriggers -Principal $principal | Out-Null
} else {
  Set-ScheduledTask -TaskName $taskName -Action $taskAction -Trigger $taskTriggers | Out-Null
}

New-PendingDrafts -Count 5
Show-ApprovalMechanics
Write-Host "Pending drafts generated: $pending"
Write-Host "Approved folder: $approved"
Write-Host "Ready-to-send folder: $ready"
