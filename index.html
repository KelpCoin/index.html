2) Supabase schema (copy/paste SQL)

Designed for high-cardinality inventory with fast lookups and price history tracking.

-- Core inventory table (100k+ rows friendly)
create table if not exists public.cards_inventory (
  id bigserial primary key,
  scryfall_id uuid not null,
  card_name text not null,
  set_code text,
  collector_number text,
  condition text not null default 'NM',
  language text not null default 'EN',
  quantity integer not null default 1,
  usd_market numeric(10,2),
  usd_source text,
  fx_usd_nzd numeric(12,6),
  nzd_list numeric(10,2),
  pricing_rule_id bigint,
  updated_at timestamptz not null default now(),
  unique (scryfall_id, condition, language)
);

create index if not exists idx_cards_scryfall on public.cards_inventory (scryfall_id);
create index if not exists idx_cards_name on public.cards_inventory (card_name);
create index if not exists idx_cards_updated on public.cards_inventory (updated_at desc);

-- Pricing rules so you can test multiple strategies
create table if not exists public.pricing_rules (
  id bigserial primary key,
  rule_name text not null,
  discount_pct numeric(5,2) not null default 0,
  buffer_pct numeric(5,2) not null default 0,
  floor_nzd numeric(10,2) not null default 0,
  round_to_cents numeric(4,2) not null default 0.05,
  active boolean not null default true,
  created_at timestamptz not null default now()
);

-- Optional: keep a full audit trail for pricing changes
create table if not exists public.price_history (
  id bigserial primary key,
  card_inventory_id bigint not null references public.cards_inventory(id) on delete cascade,
  old_nzd_list numeric(10,2),
  new_nzd_list numeric(10,2) not null,
  old_fx_usd_nzd numeric(12,6),
  new_fx_usd_nzd numeric(12,6) not null,
  changed_at timestamptz not null default now(),
  reason text
);

3) Pricing function logic

Use this SQL function from your Edge Function/batch job.

create or replace function public.calc_nzd_price(
  usd_market numeric,
  usd_nzd_rate numeric,
  discount_pct numeric,
  buffer_pct numeric,
  floor_nzd numeric default 0,
  round_to_cents numeric default 0.05
)
returns numeric
language plpgsql
as $$
declare
  raw_nzd numeric;
  adjusted numeric;
begin
  raw_nzd := usd_market * usd_nzd_rate;

  -- discount means customer pays less than pure conversion
  adjusted := raw_nzd
    * (1 - discount_pct / 100.0)
    * (1 + buffer_pct / 100.0);

  if adjusted < floor_nzd then
    adjusted := floor_nzd;
  end if;

  -- round to nearest 5 cents by default
  return round(adjusted / round_to_cents) * round_to_cents;
end;
$$;

4) Workflow blueprint

  • Matterbox scan → get scryfall_id.
  • Upsert into Supabase with quantity/condition/language.
  • Scheduled job (e.g. every hour) fetches USD card prices + current USD/NZD FX.
  • Edge Function calculates NZD list price via calc_nzd_price.
  • Store history in price_history for transparency and strategy tuning.
